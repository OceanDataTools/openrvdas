openrvdas.server.logger_manager

Manage the loggers defined by an OpenRVDAS cruise definition.

LoggerManager is the central process-control component of OpenRVDAS. It reads desired logger configurations from a ServerAPI-based data store (in-memory, SQLite or Django), runs each logger configuration in its own process via a LoggerRunner, restarts loggers that die unexpectedly, and reports logger status to a CachedDataServer.

May be invoked from the command line as, e.g.

    server/logger_manager.py         --config test/NBP1406/NBP1406_cruise.yaml         --mode monitor         --start_data_server

which will load the specified cruise definition, set it to the specified mode and start a command line console for issuing commands. Type "help" at the console for a list of commands.

In production it is typically run via supervisord with a persistent database backing it:

    server/logger_manager.py --database django --no-console         --data_server_websocket :8766

 264    def start(self):
 265        """Start the threads that make up the LoggerManager operation:
 266
 267        1. Loop to update configs from the API, check that the loggers
 268           are running what they should be (restarting any that have died
 269           unexpectedly), and send status reports to the cached data server.
 270        2. Loop to check whether the cruise definition has changed and, if
 271           so, send the updated definition to the cached data server.
 272
 273        Start threads as daemons so that they'll automatically terminate
 274        if the main thread does.
 275        """
 276        logging.info('Starting LoggerManager')
 277
 278        self.run_loggers_thread = threading.Thread(
 279            name='run_loggers_loop',
 280            target=self._run_loggers_loop, daemon=True)
 281        self.run_loggers_thread.start()
 282
 283        self.send_cruise_definition_thread = threading.Thread(
 284            name='send_cruise_definition_loop',
 285            target=self._send_cruise_definition_loop, daemon=True)
 286        self.send_cruise_definition_thread.start()

 289    def quit(self):
 290        """Exit the loops and shut down all loggers."""
 291        self.quit_flag = True
 292        with self.config_lock:
 293            for logger in list(self.logger_states):
 294                self._journal_logger_stopped(logger, reason='manager stopping')
 295                self._delete_logger(logger)
 296        self.journal.record('manager_stop', host=HOSTNAME, pid=os.getpid())
 297
 298        # Give our data server writer a chance to deliver anything still
 299        # queued (e.g. the loggers' final stderr lines) before we exit.
 300        if self.data_server_writer:
 301            self.data_server_writer.close()

 304    def update_configs(self):
 305        """Fetch the latest desired configs from the API and start/stop
 306        loggers as necessary to match them.
 307
 308        Called periodically from _run_loggers_loop(), and also registered
 309        as a callback so the API can invoke it when it knows the active
 310        configs have changed. Search for the line:
 311
 312          api.on_update(callback=logger_manager.update_configs)
 313
 314        in this file to see where.
 315        """
 316        # Get new configs in dict {logger: config_spec}
 317        logger_configs = self.api.get_logger_configs()
 318        if logger_configs:
 319            self._apply_configs(logger_configs)

 321    ############################
 322    def _apply_configs(self, configs):
 323        """Receive a new map of {logger: config} and start/stop loggers as
 324        necessary to match it.
 325        """
 326        with self.config_lock:
 327            # If we're in the process of quitting, go home - a different
 328            # thread is already shutting things down.
 329            if self.quit_flag:
 330                return
 331
 332            stale_loggers = set(self.logger_states) - set(configs)
 333            new_loggers = set(configs) - set(self.logger_states)
 334            other_loggers = set(self.logger_states) - stale_loggers - new_loggers
 335
 336            logging.debug('Stale: %s', stale_loggers)
 337            logging.debug('New: %s', new_loggers)
 338            logging.debug('Other: %s', other_loggers)
 339
 340            # If anything is about to change, record any change of mode
 341            # first, since it is usually the cause.
 342            changing = stale_loggers or new_loggers or any(
 343                configs[logger] != self.logger_states[logger].config
 344                for logger in other_loggers)
 345            if changing:
 346                self._journal_mode(self.api.get_active_mode())
 347
 348            # Find and shut down loggers that don't exist in our new configs
 349            for logger in stale_loggers:
 350                logging.info('Shutting down logger %s.', logger)
 351                self._journal_logger_stopped(logger, reason='no longer configured')
 352                self._delete_logger(logger)
 353
 354            # Add loggers that have first appeared in our new config and
 355            # start them up.
 356            for logger in new_loggers:
 357                new_config = configs[logger]
 358                logging.info('Starting new logger %s with %s.', logger,
 359                             new_config.get('name', 'no_name'))
 360                self._start_logger(logger, new_config)
 361                self.journal.record('logger_started', logger=logger,
 362                                    config=new_config.get('name'),
 363                                    config_sha256=config_hash(new_config),
 364                                    code=self.logger_states[logger].code,
 365                                    pid=self._pid(logger))
 366
 367            # For existing loggers, see whether their configs have
 368            # changed. If so, stop and restart with new config.
 369            for logger in other_loggers:
 370                new_config = configs[logger]
 371                old_config = self.logger_states[logger].config
 372                if new_config == old_config:
 373                    logging.debug('Config for %s unchanged.', logger)
 374                    continue
 375
 376                logging.info('Updating %s from %s to %s', logger,
 377                             old_config.get('name', 'no_name'),
 378                             new_config.get('name', 'no_name'))
 379                crashes = self.logger_states[logger].crash_count
 380                self._delete_logger(logger)
 381                self._start_logger(logger, new_config)
 382                entry = dict(logger=logger, **{'from': old_config.get('name')},
 383                             to=new_config.get('name'),
 384                             config_sha256=config_hash(new_config),
 385                             code=self.logger_states[logger].code,
 386                             pid=self._pid(logger))
 387                if crashes:
 388                    entry['crashes_before_change'] = crashes
 389                self.journal.record('config_changed', **entry)
 390
 391            self.active_configs = configs

 394    def _journal_mode(self, mode):
 395        """Record the active mode in the journal if it has changed. Called
 396        from both the update loop and the API's update callback, so locked."""
 397        with self.journaled_mode_lock:
 398            if mode != self.journaled_mode:
 399                self.journal.record('mode_changed', **{'from': self.journaled_mode},
 400                                    to=mode)
 401                self.journaled_mode = mode
 402
 403    def _journal_logger_stopped(self, logger, reason):
 404        """Record that a logger is being stopped, with how many times it had
 405        crashed if it was in the middle of a run of crashes."""
 406        state = self.logger_states.get(logger)
 407        if not state:
 408            return
 409        entry = dict(logger=logger, config=state.config.get('name'), reason=reason)
 410        if state.crash_count:
 411            entry['crashes_before_stop'] = state.crash_count
 412        self.journal.record('logger_stopped', **entry)
 413
 414    def _pid(self, logger):
 415        state = self.logger_states.get(logger)
 416        process = state.runner.process if state else None
 417        return process.pid if process else None

 420    def _start_logger(self, logger, config):
 421        """Create and start a LoggerRunner for the passed config. ONLY CALL
 422        THIS WITH config_lock HELD, for thread safety."""
 423        config_name = config.get('name', logger + '_config')
 424        logging.info('Called start_logger for %s: %s', logger, config_name)
 425
 426        stderr_filename = self.stderr_file_pattern.format(logger=logger)
 427
 428        # The runner captures the logger's actual stderr (fd 2) via a pipe
 429        # and relays each line back to us; we forward them to the cached
 430        # data server over our own connection, so the (possibly dying)
 431        # logger process is never responsible for delivering its own
 432        # death message.
 433        def stderr_callback(line, logger=logger):
 434            self._forward_logger_stderr(logger, line)
 435
 436        # The relay thread also tells us - via pipe EOF - when the process
 437        # has died, so we can restart it immediately instead of waiting
 438        # for the next polling check.
 439        def death_callback(runner, logger=logger):
 440            self._handle_logger_death(runner, logger)
 441
 442        runner = LoggerRunner(config=config, name=logger,
 443                              stderr_filename=stderr_filename,
 444                              stderr_max_bytes=self.stderr_max_bytes,
 445                              stderr_backup_count=self.stderr_backup_count,
 446                              stderr_callback=stderr_callback,
 447                              death_callback=death_callback,
 448                              logger_log_level=self.logger_log_level)
 449        self.logger_states[logger] = LoggerState(config=config, runner=runner)
 450        self.logger_states[logger].code = self._code_version()
 451        runner.start()
 452
 453    ############################
 454    def _delete_logger(self, logger):
 455        """Shut down the named logger and forget about it. ONLY CALL THIS
 456        WITH config_lock HELD, for thread safety."""
 457        state = self.logger_states.get(logger)
 458        if not state:
 459            logging.warning('Stale logger %s not found?!?', logger)
 460            return
 461        logging.info('Waiting for logger %s to complete', logger)
 462        state.runner.quit()
 463        del self.logger_states[logger]

 466    def _check_loggers(self):
 467        """Polling safety net: check that all the loggers we ought to be
 468        running are running, and restart any that have died unexpectedly.
 469        The primary death signal is _handle_logger_death(), called by each
 470        runner's stderr relay thread when its pipe hits EOF; this loop
 471        catches anything that slips past it.
 472        """
 473        logging.debug('Checking loggers...')
 474        with self.config_lock:
 475            for logger, state in self.logger_states.items():
 476                self._consider_restart(logger, state)
 477                self._check_recovery(logger, state)
 478
 479    ############################
 480    def _check_recovery(self, logger, state):
 481        """If a logger that has been crashing has now stayed up for
 482        min_uptime seconds, record in the journal that it has recovered, and
 483        how many times it crashed. ONLY CALL THIS WITH config_lock HELD."""
 484        if not state.crash_count or not state.runner.is_alive():
 485            return
 486        if time.time() - state.last_started < self.min_uptime:
 487            return
 488        self.journal.record('logger_recovered', logger=logger,
 489                            config=state.config.get('name'),
 490                            crashes=state.crash_count,
 491                            crashing_for_seconds=round(state.last_started -
 492                                                       state.crash_episode_start, 1),
 493                            code=state.code, pid=self._pid(logger))
 494        state.crash_count = 0
 495        state.crash_episode_start = None
 496
 497    ############################
 498    def _consider_restart(self, logger, state):
 499        """If the passed logger is unexpectedly dead, restart it - unless it
 500        has died max_tries times without staying up at least min_uptime
 501        seconds, in which case declare it failed and stop trying. ONLY CALL
 502        THIS WITH config_lock HELD, for thread safety.
 503        """
 504        runner = state.runner
 505        if not runner.is_runnable():
 506            return
 507        if runner.is_alive():
 508            return
 509        if runner.is_failed():
 510            return
 511
 512        # If we're here, runner is runnable, is not running, and hasn't
 513        # yet been labeled as a failed logger.
 514        logging.warning('%s unexpectedly dead.', logger)
 515
 516        # Journal only the first crash of a run of crashes; the rest are
 517        # counted, and reported when the logger recovers, fails or stops.
 518        if not state.crash_count:
 519            state.crash_episode_start = time.time()
 520            exit_code = runner.process.returncode if runner.process else None
 521            self.journal.record('logger_crashed', logger=logger,
 522                                config=state.config.get('name'), exit_code=exit_code,
 523                                code=state.code)
 524        state.crash_count += 1
 525
 526        # How long was it up? If long enough, give it a clean slate.
 527        if time.time() - state.last_started < self.min_uptime:
 528            state.restart_count += 1
 529        else:
 530            state.restart_count = 0
 531
 532        # If we've restarted too many times recently, declare the logger
 533        # failed and move on. max_tries of zero means never stop retrying.
 534        if self.max_tries and state.restart_count >= self.max_tries:
 535            runner.failed = True
 536            logging.warning('%s has failed %s times; not restarting',
 537                            logger, state.restart_count)
 538            self.journal.record('logger_failed', logger=logger,
 539                                config=state.config.get('name'),
 540                                crashes=state.crash_count)
 541            return
 542
 543        # If here, we're going to try restarting.
 544        logging.info('%s - restarting', logger)
 545        state.last_started = time.time()
 546        state.code = self._code_version()
 547        runner.start()
 548

 550    def _handle_logger_death(self, runner, logger):
 551        """Called - from the dead runner's stderr relay thread - when a
 552        logger process's stderr pipe hits EOF, i.e. the process has exited
 553        and its last words have been delivered. Restart it (or declare it
 554        failed) right away rather than waiting for the next polling check.
 555        """
 556        if self.quit_flag or runner.quit_flag:
 557            return  # deliberate shutdown, not a death
 558
 559        # Acquire the lock with a timeout: a manager thread holding it may
 560        # be in runner.quit(), joining the very relay thread we're running
 561        # in. Re-checking the quit flags lets us bail out instead of
 562        # stalling that join.
 563        while not self.config_lock.acquire(timeout=0.5):
 564            if self.quit_flag or runner.quit_flag:
 565                return
 566        try:
 567            state = self.logger_states.get(logger)
 568            if not state or state.runner is not runner:
 569                return  # stale notification; logger was reconfigured
 570            self._consider_restart(logger, state)
 571        finally:
 572            self.config_lock.release()

 575    def get_status(self):
 576        """Return a dict of the current config name and current run status of
 577        each logger in the form, e.g.:
 578
 579        {'s330': {'config':'s330->net', 'status':'RUNNING'},
 580         'gyr1': {'config':'gyr1->file', 'status':'FAILED'},
 581        }
 582
 583        Possible status are EXITED, RUNNING, FAILED and STARTING. EXITED
 584        is the status when a logger is not 'runnable' - e.g. the 'off'
 585        config. STARTING is used when a runner is runnable but is not
 586        running and is not FAILED - i.e. we haven't given up on it.
 587        """
 588        logger_status = {}
 589        with self.config_lock:
 590            for logger, state in self.logger_states.items():
 591                config_name = state.config.get('name', 'no name')
 592
 593                if not state.runner.is_runnable():
 594                    status = 'EXITED'
 595                elif state.runner.is_alive():
 596                    status = 'RUNNING'
 597                elif state.runner.is_failed():
 598                    status = 'FAILED'
 599                else:
 600                    status = 'STARTING'
 601                logger_status[logger] = {'config': config_name, 'status': status}
 602        return logger_status

 605    def _run_loggers_loop(self):
 606        """Main supervision loop: update configs from the API, restart any
 607        loggers that have died unexpectedly, and send status reports to the
 608        cached data server.
 609        """

 611            try:
 612                self.update_configs()
 613                self._check_loggers()
 614                self._send_status()

 616                # Catch broadly: a transient API error (e.g. SQLAlchemy
 617                # NoResultFound when no cruise is loaded yet) must not kill
 618                # this thread; we recover on the next iteration. See
 619                # upstream #562.
 620                logging.warning('Error in logger update loop: %s', e)
 621            time.sleep(self.interval)

 682    def _journal_definition_loaded(self, loaded_time):
 683        """Record a newly loaded cruise definition, with a hash of its file
 684        so the journal can be matched to a snapshot of it."""
 685        # Some data stores report the cruise id and filename at the top
 686        # level of the configuration, others under its 'cruise' key.
 687        cruise_section = self.cruise.get('cruise') or {}
 688        cruise_id = self.cruise.get('id') or cruise_section.get('id')
 689        filename = self.cruise_filename or cruise_section.get('config_filename')
 690
 691        # The same load is reported both by the API's callback and by the
 692        # loop that watches for new definitions; record it once.
 693        if (cruise_id, loaded_time) == self.journaled_definition:
 694            return
 695        self.journaled_definition = (cruise_id, loaded_time)
 696
 697        # Data stores give the load time as a datetime, naive UTC or aware.
 698        if isinstance(loaded_time, datetime.datetime):
 699            if loaded_time.tzinfo is not None:
 700                loaded_time = loaded_time.astimezone(datetime.timezone.utc).replace(tzinfo=None)
 701            loaded_time = loaded_time.strftime('%Y-%m-%dT%H:%M:%S.%f')[:-3] + 'Z'
 702
 703        file_sha256 = None
 704        if filename:
 705            try:
 706                with open(filename, 'rb') as f:
 707                    file_sha256 = hashlib.sha256(f.read()).hexdigest()
 708            except OSError:
 709                pass
 710
 711        # The hash of every file the definition is built from - its includes,
 712        # device definitions and so on - so that a snapshot can tell exactly
 713        # which have changed since, and so we can watch them.
 714        files = self._definition_file_hashes(filename)
 715        self.definition_files = {path: (self._mtime(path), sha256)
 716                                 for path, sha256 in (files or {}).items()}
 717        self.definition_filename = filename
 718
 719        self.journal.record('definition_loaded', cruise_id=cruise_id, filename=filename,
 720                            loaded_time=loaded_time, file_sha256=file_sha256, files=files)
 721        self._take_snapshot(filename, reason='definition loaded')

 743    def _definition_file_hashes(self, filename):
 744        """{path: sha256} of the files a definition is built from, or None
 745        if they can't be determined. Never raises."""

 758    def _mtime(path):
 759        try:
 760            return os.path.getmtime(path)
 761        except OSError:

 765    def _check_definition_files(self):
 766        """If any file the loaded definition was built from has changed,
 767        record which in the journal, and take a snapshot if we're taking
 768        them. Cheap when nothing has changed: one stat per file."""
 769        changed = {}
 770        for path, (mtime, sha256) in list(self.definition_files.items()):
 771            new_mtime = self._mtime(path)
 772            if new_mtime == mtime:
 773                continue
 774            new_sha256 = None
 775            if new_mtime is not None:
 776                try:
 777                    with open(path, 'rb') as f:
 778                        new_sha256 = hashlib.sha256(f.read()).hexdigest()
 779                except OSError:
 780                    pass
 781            self.definition_files[path] = (new_mtime, new_sha256)
 782            if new_sha256 != sha256:   # touched but unchanged isn't a change
 783                changed[path] = new_sha256
 784        if changed:
 785            self.journal.record('definition_files_changed', filename=self.definition_filename,
 786                                files=changed)
 787            self._take_snapshot(self.definition_filename, reason='definition files changed')

 790    def _take_snapshot(self, filename, reason):
 791        """If we're taking snapshots, take one of the definition in filename,
 792        in the background, and record it in the journal."""
 793        if not self.snapshot_dir or not filename:
 794            return
 795
 796        def snapshot():
 797            try:
 798                from server.config_snapshot import take_snapshot
 799                path = take_snapshot(filename, output_dir=self.snapshot_dir,
 800                                     journal=self.journal.filebase)
 801                self.journal.record('snapshot_taken', reason=reason, path=path)
 802            except Exception as e:  # noqa: BLE001 - must not stop the manager
 803                logging.warning('Could not take snapshot of %s: %s', filename, e)
 804                self.journal.record('snapshot_failed', reason=reason, error=str(e))
 805
 806        threading.Thread(target=snapshot, name='snapshot', daemon=True).start()

 809    def _send_cruise_definition_loop(self):
 810        """Iteratively check whether the cruise definition in the data store
 811        (or its source file) has changed. If the file has changed, send a
 812        notification to the console so it can ask whether the user wants to
 813        reload. If the data store has a definition with a newer timestamp,
 814        rebuild our local maps and send the updated definition to the
 815        cached data server.
 816        """
 817        last_loaded_timestamp = 0
 818
 819        while not self.quit_flag:
 820            try:
 821                self.cruise = self.api.get_configuration()  # a Cruise object
 822                if not self.cruise:
 823                    logging.info('No cruise definition found in API')
 824                    time.sleep(self.interval * 2)
 825                    continue
 826                self.cruise_filename = self.cruise.get('config_filename')
 827                loaded_time = self.cruise.get('loaded_time')
 828                self.cruise_loaded_time = datetime.datetime.timestamp(loaded_time)
 829
 830                # Has cruise definition file changed since we loaded it? If so,
 831                # send a notification to console so it can ask if user wants to
 832                # reload.
 833                if self.cruise_filename:
 834                    try:
 835                        mtime = os.path.getmtime(self.cruise_filename)
 836                        if mtime > self.cruise_loaded_time:
 837                            logging.debug('Cruise file timestamp changed!')
 838                            self._write_record_to_data_server('status:file_update', mtime)
 839                    except FileNotFoundError:
 840                        logging.debug('Cruise file "%s" has disappeared?', self.cruise_filename)
 841
 842                # Has any file the definition was built from changed?
 843                self._check_definition_files()
 844
 845                # Does database have a cruise definition with a newer timestamp?
 846                # Means user loaded/reloaded definition. Update our maps to
 847                # reflect the new values and send an updated cruise_definition
 848                # to the console.
 849                if self.cruise_loaded_time > last_loaded_timestamp:
 850                    last_loaded_timestamp = self.cruise_loaded_time
 851                    logging.info('New cruise definition detected - rebuilding maps.')
 852                    self.load_definition_from_api()
 853
 854            except Exception as e:
 855                logging.warning('Error checking cruise definition: %s', e)
 856
 857            # Whether or not we've sent an update, sleep
 858            time.sleep(self.interval * 2)

 884class AnotherManagerRunning(Exception):
 885    """Raised by acquire_lock() when another logger manager holds the lock."""
 886
 887    def __init__(self, running_pid):
 888        super().__init__(running_pid)
 889        self.running_pid = running_pid
 890
 891
 892def acquire_lock(lock_file):
 893    """Take an exclusive lock on lock_file, so that only one logger manager
 894    runs on this host at a time, and write our pid into it (#686).
 895
 896    Returns the open lock file, which must be kept open: the lock lasts as
 897    long as it does, and the operating system releases it when this process
 898    exits, however it exits, so a crash leaves no stale lock to clear. Returns
 899    None, with a warning, if the lock can't be taken for any reason other than
 900    another manager holding it - an unwritable directory, say - so that the
 901    manager still runs. Raises AnotherManagerRunning, with the pid recorded in
 902    the file, if another manager holds it.
 903    """
 904    try:
 905        os.makedirs(os.path.dirname(os.path.abspath(lock_file)), exist_ok=True)
 906        lock = open(lock_file, 'a+')  # not 'w': that would empty a held lock's pid
 907    except OSError as e:
 908        logging.warning('Could not open lock file %s (%s), so not checking for '
 909                        'another logger manager on this host.', lock_file, e)
 910        return None
 911    try:
 912        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
 913    except BlockingIOError:
 914        lock.seek(0)
 915        running_pid = lock.read().strip() or 'unknown'
 916        lock.close()
 917        raise AnotherManagerRunning(running_pid)
 918    except OSError as e:
 919        logging.warning('Could not lock %s (%s), so not checking for another '
 920                        'logger manager on this host.', lock_file, e)
 921        lock.close()
 922        return None
 923    lock.seek(0)
 924    lock.truncate()
 925    lock.write(f'{os.getpid()}\n')
 926    lock.flush()
 927    return lock

481    def _check_recovery(self, logger, state):
482        """If a logger that has been crashing has now stayed up for
483        min_uptime seconds, record in the journal that it has recovered, and
484        how many times it crashed. ONLY CALL THIS WITH config_lock HELD."""
485        if not state.crash_count or not state.runner.is_alive():
486            return
487        if time.time() - state.last_started < self.min_uptime:
488            return
489        self.journal.record('logger_recovered', logger=logger,
490                            config=state.config.get('name'),
491                            crashes=state.crash_count,
492                            crashing_for_seconds=round(state.last_started -
493                                                       state.crash_episode_start, 1),
494                            code=state.code, pid=self._pid(logger))
495        state.crash_count = 0

499    def _consider_restart(self, logger, state):
500        """If the passed logger is unexpectedly dead, restart it - unless it
501        has died max_tries times without staying up at least min_uptime
502        seconds, in which case declare it failed and stop trying. ONLY CALL
503        THIS WITH config_lock HELD, for thread safety.
504        """
505        runner = state.runner
506        if not runner.is_runnable():
507            return
508        if runner.is_alive():
509            return
510        if runner.is_failed():
511            return
512
513        # If we're here, runner is runnable, is not running, and hasn't
514        # yet been labeled as a failed logger.
515        logging.warning('%s unexpectedly dead.', logger)
516
517        # Journal only the first crash of a run of crashes; the rest are
518        # counted, and reported when the logger recovers, fails or stops.
519        if not state.crash_count:
520            state.crash_episode_start = time.time()
521            exit_code = runner.process.returncode if runner.process else None
522            self.journal.record('logger_crashed', logger=logger,
523                                config=state.config.get('name'), exit_code=exit_code,
524                                code=state.code)
525        state.crash_count += 1
526
527        # How long was it up? If long enough, give it a clean slate.
528        if time.time() - state.last_started < self.min_uptime:
529            state.restart_count += 1
530        else:
531            state.restart_count = 0
532
533        # If we've restarted too many times recently, declare the logger
534        # failed and move on. max_tries of zero means never stop retrying.
535        if self.max_tries and state.restart_count >= self.max_tries:
536            runner.failed = True
537            logging.warning('%s has failed %s times; not restarting',
538                            logger, state.restart_count)
539            self.journal.record('logger_failed', logger=logger,
540                                config=state.config.get('name'),
541                                crashes=state.crash_count)
542            return
543
544        # If here, we're going to try restarting.
545        logging.info('%s - restarting', logger)
546        state.last_started = time.time()
547        state.code = self._code_version()
548        runner.start()

551    def _handle_logger_death(self, runner, logger):
552        """Called - from the dead runner's stderr relay thread - when a
553        logger process's stderr pipe hits EOF, i.e. the process has exited
554        and its last words have been delivered. Restart it (or declare it
555        failed) right away rather than waiting for the next polling check.
556        """
557        if self.quit_flag or runner.quit_flag:
558            return  # deliberate shutdown, not a death
559
560        # Acquire the lock with a timeout: a manager thread holding it may
561        # be in runner.quit(), joining the very relay thread we're running
562        # in. Re-checking the quit flags lets us bail out instead of
563        # stalling that join.
564        while not self.config_lock.acquire(timeout=0.5):
565            if self.quit_flag or runner.quit_flag:
566                return
567        try:
568            state = self.logger_states.get(logger)
569            if not state or state.runner is not runner:
570                return  # stale notification; logger was reconfigured
571            self._consider_restart(logger, state)
572        finally:
573            self.config_lock.release()

576    def get_status(self):
577        """Return a dict of the current config name and current run status of
578        each logger in the form, e.g.:
579
580        {'s330': {'config':'s330->net', 'status':'RUNNING'},
581         'gyr1': {'config':'gyr1->file', 'status':'FAILED'},
582        }
583
584        Possible status are EXITED, RUNNING, FAILED and STARTING. EXITED
585        is the status when a logger is not 'runnable' - e.g. the 'off'
586        config. STARTING is used when a runner is runnable but is not
587        running and is not FAILED - i.e. we haven't given up on it.
588        """
589        logger_status = {}
590        with self.config_lock:
591            for logger, state in self.logger_states.items():
592                config_name = state.config.get('name', 'no name')
593
594                if not state.runner.is_runnable():
595                    status = 'EXITED'
596                elif state.runner.is_alive():
597                    status = 'RUNNING'
598                elif state.runner.is_failed():
599                    status = 'FAILED'
600                else:
601                    status = 'STARTING'
602                logger_status[logger] = {'config': config_name, 'status': status}
603        return logger_status

683    def _journal_definition_loaded(self, loaded_time):
684        """Record a newly loaded cruise definition, with a hash of its file
685        so the journal can be matched to a snapshot of it."""
686        # Some data stores report the cruise id and filename at the top
687        # level of the configuration, others under its 'cruise' key.
688        cruise_section = self.cruise.get('cruise') or {}
689        cruise_id = self.cruise.get('id') or cruise_section.get('id')
690        filename = self.cruise_filename or cruise_section.get('config_filename')
691
692        # The same load is reported both by the API's callback and by the
693        # loop that watches for new definitions; record it once.
694        if (cruise_id, loaded_time) == self.journaled_definition:
695            return
696        self.journaled_definition = (cruise_id, loaded_time)
697
698        # Data stores give the load time as a datetime, naive UTC or aware.
699        if isinstance(loaded_time, datetime.datetime):
700            if loaded_time.tzinfo is not None:
701                loaded_time = loaded_time.astimezone(datetime.timezone.utc).replace(tzinfo=None)
702            loaded_time = loaded_time.strftime('%Y-%m-%dT%H:%M:%S.%f')[:-3] + 'Z'
703
704        file_sha256 = None
705        if filename:
706            try:
707                with open(filename, 'rb') as f:
708                    file_sha256 = hashlib.sha256(f.read()).hexdigest()
709            except OSError:
710                pass
711
712        # The hash of every file the definition is built from - its includes,
713        # device definitions and so on - so that a snapshot can tell exactly
714        # which have changed since, and so we can watch them.
715        files = self._definition_file_hashes(filename)
716        self.definition_files = {path: (self._mtime(path), sha256)
717                                 for path, sha256 in (files or {}).items()}
718        self.definition_filename = filename
719
720        self.journal.record('definition_loaded', cruise_id=cruise_id, filename=filename,
721                            loaded_time=loaded_time, file_sha256=file_sha256, files=files)
722        self._take_snapshot(filename, reason='definition loaded')

766    def _check_definition_files(self):
767        """If any file the loaded definition was built from has changed,
768        record which in the journal, and take a snapshot if we're taking
769        them. Cheap when nothing has changed: one stat per file."""
770        changed = {}
771        for path, (mtime, sha256) in list(self.definition_files.items()):
772            new_mtime = self._mtime(path)
773            if new_mtime == mtime:
774                continue
775            new_sha256 = None
776            if new_mtime is not None:
777                try:
778                    with open(path, 'rb') as f:
779                        new_sha256 = hashlib.sha256(f.read()).hexdigest()
780                except OSError:
781                    pass
782            self.definition_files[path] = (new_mtime, new_sha256)
783            if new_sha256 != sha256:   # touched but unchanged isn't a change
784                changed[path] = new_sha256
785        if changed:
786            self.journal.record('definition_files_changed', filename=self.definition_filename,
787                                files=changed)
788            self._take_snapshot(self.definition_filename, reason='definition files changed')

893def acquire_lock(lock_file):
894    """Take an exclusive lock on lock_file, so that only one logger manager
895    runs on this host at a time, and write our pid into it (#686).
896
897    Returns the open lock file, which must be kept open: the lock lasts as
898    long as it does, and the operating system releases it when this process
899    exits, however it exits, so a crash leaves no stale lock to clear. Returns
900    None, with a warning, if the lock can't be taken for any reason other than
901    another manager holding it - an unwritable directory, say - so that the
902    manager still runs. Raises AnotherManagerRunning, with the pid recorded in
903    the file, if another manager holds it.
904    """
905    try:
906        os.makedirs(os.path.dirname(os.path.abspath(lock_file)), exist_ok=True)
907        lock = open(lock_file, 'a+')  # not 'w': that would empty a held lock's pid
908    except OSError as e:
909        logging.warning('Could not open lock file %s (%s), so not checking for '
910                        'another logger manager on this host.', lock_file, e)
911        return None
912    try:
913        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
914    except BlockingIOError:
915        lock.seek(0)
916        running_pid = lock.read().strip() or 'unknown'
917        lock.close()
918        raise AnotherManagerRunning(running_pid)
919    except OSError as e:
920        logging.warning('Could not lock %s (%s), so not checking for another '
921                        'logger manager on this host.', lock_file, e)
922        lock.close()
923        return None
924    lock.seek(0)
925    lock.truncate()
926    lock.write(f'{os.getpid()}\n')
927    lock.flush()
928    return lock
